
function(add_module name)
    nanobind_add_module(${name} ${ARGN})
    nanobind_custom_build(${name} ${CMAKE_SOURCE_DIR}/bin)
    target_link_libraries(${name} PRIVATE annobind)
endfunction()

add_module(ab_test ab_test.cpp)
target_include_directories(ab_test PRIVATE ${CMAKE_SOURCE_DIR}/include)
